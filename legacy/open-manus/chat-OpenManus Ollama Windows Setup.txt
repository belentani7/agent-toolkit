## 🐛 Troubleshooting

### Error: "ModuleNotFoundError"
```cmd
venv\Scripts\activate
pip install -r requirements.txt --force-reinstall
```

### Error: "Connection refused" en localhost:11434
```cmd
# Verifica que Ollama esté corriendo
ollama serve

# O reinicia el servicio
taskkill /F /IM ollama.exe
ollama serve
```

### Error: "Out of memory"
- Cierra aplicaciones que consuman RAM
- Usa un modelo más pequeño: `ollama pull qwen2.5:3b`
- Aumenta `max_tokens` en config.toml

### La Web UI no carga
```cmd
# Verifica que el puerto 8000 esté libre
netstat -ano | findstr :8000

# Usa otro puerto editando web_run.py
python web_run.py --port 8080
```

### El agente no ejecuta herramientas
- Verifica que el modelo soporte tool calling
- Reinicia con `python main.py --verbose`
- Consulta los logs en `logs/`

### 4. **setup_windows.bat** (Instalador automático)

```batch
@echo off
setlocal EnableDelayedExpansion

echo ===================================================
echo   OpenManus + Ollama Installer para Windows 11
echo ===================================================
echo.

:: Verificar Python
echo [1/6] Verificando Python...
python --version >nul 2>&1
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Python no está instalado o no está en el PATH
    echo Por favor instala Python 3.11 o 3.12 desde https://www.python.org/downloads/
    echo Asegúrate de marcar "Add Python to PATH" durante la instalación
    pause
    exit /b 1
)

for /f "tokens=2" %%i in ('python --version 2^>^&1') do set PYTHON_VERSION=%%i
echo [OK] Python %PYTHON_VERSION% detectado

:: Verificar Git
echo [2/6] Verificando Git...
git --version >nul 2>&1
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Git no está instalado
    echo Por favor instala Git desde https://git-scm.com/download/win
    pause
    exit /b 1
)
echo [OK] Git detectado

:: Verificar Ollama
echo [3/6] Verificando Ollama...
where ollama >nul 2>&1
if %ERRORLEVEL% NEQ 0 (
    echo [WARNING] Ollama no está instalado
    echo Descargando Ollama...
    powershell -Command "Invoke-WebRequest -Uri 'https://ollama.com/download/OllamaSetup.exe' -OutFile 'OllamaSetup.exe'"
    echo Instalando Ollama...
    start /wait OllamaSetup.exe
    del OllamaSetup.exe
    echo [OK] Ollama instalado
) else (
    echo [OK] Ollama detectado
)

:: Crear entorno virtual
echo [4/6] Creando entorno virtual de Python...
if exist "venv" (
    echo [INFO] Entorno virtual ya existe, reutilizando...
) else (
    python -m venv venv
    if %ERRORLEVEL% NEQ 0 (
        echo [ERROR] No se pudo crear el entorno virtual
        pause
        exit /b 1
    )
)
echo [OK] Entorno virtual creado

:: Instalar dependencias
echo [5/6] Instalando dependencias con uv...
call venv\Scripts\activate

python -m pip install --upgrade pip --quiet
pip install uv --quiet

if not exist "requirements.txt" (
    echo [ERROR] requirements.txt no encontrado
    pause
    exit /b 1
)

uv pip install -r requirements.txt
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Error instalando dependencias
    echo Intentando con pip estándar...
    pip install -r requirements.txt
)
echo [OK] Dependencias instaladas

:: Instalar Playwright browsers
echo [INFO] Instalando navegadores para Playwright...
python -m playwright install chromium

:: Configurar Ollama
echo [6/6] Configurando Ollama y descargando modelo...
if not exist "config" mkdir config
if not exist "config\config.toml" (
    if exist "config\config.example-model-ollama.toml" (
        copy config\config.example-model-ollama.toml config\config.toml
        echo [OK] Archivo de configuración creado
    )
) else (
    echo [INFO] config.toml ya existe, no se sobrescribirá
)

:: Descargar modelo
echo.
echo Descargando modelo qwen2.5:7b (esto puede tardar varios minutos)...
ollama pull qwen2.5:7b

if %ERRORLEVEL% NEQ 0 (
    echo [WARNING] Error descargando modelo, intenta manualmente:
    echo ollama pull qwen2.5:7b
)

echo.
echo ===================================================
echo   Instalación Completada Exitosamente!
echo ===================================================
echo.
echo Próximos pasos:
echo.
echo 1. Activa el entorno virtual:
echo    venv\Scripts\activate
echo.
echo 2. Inicia la interfaz web:
echo    python web_run.py
echo.
echo 3. Abre tu navegador en: http://localhost:8000
echo.
echo Para modo CLI usa: python main.py
echo Para modo MCP usa: python run_mcp.py
echo.
echo ===================================================
pause
```

### 5. **uninstall.bat** (Desinstalador)

```batch
@echo off
echo ===================================================
echo   Desinstalador de OpenManus + Ollama
echo ===================================================
echo.

set /p CONFIRM="¿Estás seguro de que deseas desinstalar? (S/N): "
if /i not "%CONFIRM%"=="S" (
    echo Desinstalación cancelada.
    pause
    exit /b 0
)

echo [1/3] Eliminando entorno virtual...
if exist "venv" (
    rmdir /s /q venv
    echo [OK] Entorno virtual eliminado
) else (
    echo [INFO] No se encontró entorno virtual
)

echo [2/3] Eliminando archivos de configuración...
if exist "config\config.toml" (
    del config\config.toml
    echo [OK] config.toml eliminado
)

echo [3/3] Limpiando archivos temporales...
if exist "logs" rmdir /s /q logs
if exist "workspace" rmdir /s /q workspace
if exist "__pycache__" rmdir /s /q __pycache__

echo.
echo ===================================================
echo   Desinstalación Completada!
echo ===================================================
echo.
echo Nota: Ollama y los modelos descargados NO fueron eliminados.
echo Para desinstalar Ollama completamente, usa el desinstalador de Windows.
echo.
pause
```

### 7. **start_web.bat** (Script rápido para iniciar Web UI)

```batch
@echo off
echo Iniciando OpenManus Web UI...
echo.

if not exist "venv" (
    echo [ERROR] Entorno virtual no encontrado
    echo Ejecuta setup_windows.bat primero
    pause
    exit /b 1
)

call venv\Scripts\activate

:: Verificar Ollama
tasklist | findstr "ollama" >nul
if %ERRORLEVEL% NEQ 0 (
    echo [INFO] Iniciando Ollama...
    start /B ollama serve
    timeout /t 5 /nobreak >nul
)

echo Abriendo Web UI en http://localhost:8000
echo.
echo Presiona Ctrl+C para detener
echo.

python web_run.py

pause
```

### 9. **check_system.bat** (Verificador del sistema)

```batch
@echo off
echo ===================================================
echo   Verificador del Sistema - OpenManus + Ollama
echo ===================================================
echo.

:: Check Python
echo [1/5] Verificando Python...
python --version 2>nul
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Python no está instalado
) else (
    echo [OK] Python detectado
)
echo.

:: Check Git
echo [2/5] Verificando Git...
git --version 2>nul
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Git no está instalado
) else (
    echo [OK] Git detectado
)
echo.

:: Check Ollama
echo [3/5] Verificando Ollama...
where ollama >nul 2>&1
if %ERRORLEVEL% NEQ 0 (
    echo [ERROR] Ollama no está instalado
) else (
    echo [OK] Ollama detectado
    echo Modelos disponibles:
    ollama list
)
echo.

:: Check Virtual Environment
echo [4/5] Verificando entorno virtual...
if exist "venv" (
    echo [OK] Entorno virtual encontrado
    call venv\Scripts\activate
    pip list | findstr "pydantic openai"
) else (
    echo [ERROR] Entorno virtual no encontrado
)
echo.

:: Check Configuration
echo [5/5] Verificando configuración...
if exist "config\config.toml" (
    echo [OK] config.toml encontrado
) else (
    echo [WARNING] config.toml no encontrado
)
echo.

echo ===================================================
echo   Verificación Completada
echo ===================================================
pause
```
