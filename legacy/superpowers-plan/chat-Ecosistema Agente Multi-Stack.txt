```sql
RETURNS TRIGGER AS $$
DECLARE
    last_hash VARCHAR(64);
BEGIN
    -- Obtener el hash de la última auditoría de este agente
    SELECT payload_hash INTO last_hash
    FROM public.legal_audit
    WHERE agent_id = NEW.agent_id
    ORDER BY executed_at DESC
    LIMIT 1;

    -- El payload_hash debe ser calculado en la aplicación antes del INSERT, 
    -- pero podemos validar que no sea nulo.
    RETURN NEW;
END;

    match_threshold float,
    match_count int
)
RETURNS TABLE (

)

    FROM episodic_memory
```

### C1. Pipeline de Análisis Estático y AST (Abstract Syntax Tree)
No se indexa texto plano. Se extrae la estructura semántica del código para que el agente entienda *qué hace* el código, no solo *cómo se ve*.
1. **Herramienta:** `Tree-sitter` (bindings de Node.js o Python).
2. **Proceso:**
   - Clonar repositorios objetivo en un directorio `/workspace/ingestion`.
   - Recorrer archivos `.ts`, `.py`, `.go`, `.java`.
   - Generar un grafo de dependencias: `Archivo A` importa `Función B` de `Archivo C`.
   - Extraer: Nombres de funciones, firmas, docstrings y variables de entorno referenciadas.

### C2. Estrategia de Chunking Semántico y Embeddings
Dividir el código en fragmentos que tengan sentido para un LLM, no por límite de caracteres.
- **Chunking:** Usar `langchain.text_splitter.RecursiveCharacterTextSplitter` con separadores específicos de lenguaje (`\nfunction`, `\nclass`, `\n#`).
- **Metadatos enriquecidos:** Cada chunk debe llevar: `{ "repo": "agentguard", "file": "src/router.ts", "line_start": 45, "line_end": 80, "language": "typescript" }`.
- **Embedding:** Usar `nomic-embed-text` (local, gratuito, 768 dimensiones) o `text-embedding-3-small` (OpenAI, 1536 dimensiones).
- **Inserción:** Batch insert en Supabase `episodic_memory` con el embedding generado.

```sql
      AND (filter_repo IS NULL OR episodic_memory.repo_context = filter_repo)
    ORDER BY similarity DESC
    LIMIT match_count;
END;
$$;
```

```python
import os

def generate_mock_embedding(text: str) -> list[float]:
    """
    REPLACE THIS: Llamar a OpenAI API o modelo local (nomic-embed-text)
    """
    import random
    return [random.uniform(-1, 1) for _ in range(768)]

def chunk_code(content: str, chunk_size: int = 1000) -> list[str]:
    """Divide el código en chunks por líneas para mantener contexto sintáctico."""
    lines = content.split('\n')
    chunks = []
    current_chunk = []
    current_length = 0

    for line in lines:
        current_chunk.append(line)
        current_length += len(line)
        if current_length >= chunk_size:
            chunks.append('\n'.join(current_chunk))
            current_chunk = []
            current_length = 0
    
    if current_chunk:
        chunks.append('\n'.join(current_chunk))
    return chunks

def ingest_repository(repo_name: str):
    repo_path = Path(WORKSPACE_DIR) / repo_name
    if not repo_path.exists():
        print(f"[OMEGA] Repositorio no encontrado: {repo_name}")
        return

    print(f"[OMEGA] Iniciando ingesta de: {repo_name}")
    files_processed = 0

    for ext in CODE_EXTENSIONS:
        for file_path in glob.glob(f"{repo_path}/**/*{ext}", recursive=True):
            # Ignorar node_modules, venv, etc.
            if any(ignore in file_path for ignore in ['node_modules', 'venv', '.git', 'dist', 'build']):
                continue

                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                chunks = chunk_code(content)
                relative_path = os.path.relpath(file_path, repo_path)

                for i, chunk in enumerate(chunks):

                        "repo_context": repo_name,
                        "content": chunk,

                        "metadata": {
                            "file_path": relative_path,
                            "chunk_index": i,
                            "total_chunks": len(chunks),
                            "file_hash": file_hash
                        }
                    }).execute()
                
                files_processed += 1
            except Exception as e:
                print(f"[OMEGA] Error procesando {file_path}: {e}")

    print(f"[OMEGA] Ingesta completada para {repo_name}. Archivos procesados: {files_processed}")

if __name__ == "__main__":
```

### 3.2. Supabase Edge Function para Webhooks de GitHub

```typescript
// supabase/functions/github-webhook/index.ts
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
import { createHmac } from "https://deno.land/std@0.168.0/node/crypto.ts";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type, x-hub-signature-256",
};

serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });

  // 1. Validar firma HMAC (Skill 36: webhook_signature_verification)
  const signature = req.headers.get("x-hub-signature-256");
  const body = await req.text();
  const secret = Deno.env.get("GITHUB_WEBHOOK_SECRET");
  
  const hmac = createHmac("sha256", secret!);
  hmac.update(body);
  const expectedSignature = `sha256=${hmac.digest("hex")}`;

  if (signature !== expectedSignature) {
    return new Response("Unauthorized", { status: 401, headers: corsHeaders });
  }

  const payload = JSON.parse(body);
  const repoName = payload.repository.name; // Ej: "agentguard"
  const action = payload.action;

  if (action === "created" || action === "modified") {
    const commits = payload.commits || [];
    const supabase = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);

    for (const commit of commits) {
      for (const file of (commit.added || []).concat(commit.modified || [])) {
        if (file.endsWith('.ts') || file.endsWith('.py') || file.endsWith('.go')) {
          // Disparar proceso de re-embedding asíncrono (aquí simplificado)
          console.log(`[WEBHOOK] Programando re-indexación de ${file} en ${repoName}`);
          
          await supabase.from("legal_audit").insert({
            action_type: "GITHUB_WEBHOOK_TRIGGER",
            target_resource: `${repoName}/${file}`,
            payload_hash: commit.id
          });
        }
      }
    }
  }

  return new Response(JSON.stringify({ status: "processed" }), { headers: corsHeaders });
});
```

```sql
CREATE INDEX idx_docs_embedding ON documents 
USING hnsw (embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);

-- ─── TABLA: USO Y MÉTRICAS ───
CREATE TABLE usage_metrics (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    org_id UUID REFERENCES organizations(id) ON DELETE CASCADE,
    event_type TEXT NOT NULL, -- 'api_call', 'doc_generated', 'embedding_created'
    event_data JSONB DEFAULT '{}',
    tokens_used INTEGER DEFAULT 0,
    cost_cents INTEGER DEFAULT 0, -- Coste real en centésimos
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_usage_user ON usage_metrics(user_id);
CREATE INDEX idx_usage_org ON usage_metrics(org_id);
CREATE INDEX idx_usage_type ON usage_metrics(event_type);
CREATE INDEX idx_usage_created ON usage_metrics(created_at DESC);

-- Particionado por mes para performance
CREATE TABLE usage_metrics_partitioned (
    LIKE usage_metrics INCLUDING ALL
) PARTITION BY RANGE (created_at);

-- Crear particiones para próximos 12 meses
DO $$
DECLARE
    start_date DATE := DATE_TRUNC('month', CURRENT_DATE);
    end_date DATE;
    partition_name TEXT;
BEGIN
    FOR i IN 0..11 LOOP
        end_date := start_date + INTERVAL '1 month';
        partition_name := 'usage_metrics_' || TO_CHAR(start_date, 'YYYY_MM');
        EXECUTE format('CREATE TABLE IF NOT EXISTS %I PARTITION OF usage_metrics_partitioned FOR VALUES FROM (%L) TO (%L)',
            partition_name, start_date, end_date);
        start_date := end_date;
    END LOOP;
END $$;

-- ─── TABLA: AUDITORÍA INMUTABLE (hash chain) ───
CREATE TABLE audit_log (
    id BIGSERIAL PRIMARY KEY,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    org_id UUID REFERENCES organizations(id) ON DELETE SET NULL,
    action audit_action NOT NULL,
    resource_type TEXT NOT NULL,
    resource_id UUID,
    ip_address INET,
    user_agent TEXT,
    metadata JSONB DEFAULT '{}',
    payload_hash TEXT NOT NULL, -- SHA-256 del payload
    previous_hash TEXT, -- Hash del registro anterior (cadena)
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Función para calcular hash chain
CREATE OR REPLACE FUNCTION calculate_audit_hash(
    p_action audit_action,
    p_resource_type TEXT,
    p_resource_id UUID,
    p_metadata JSONB,
    p_previous_hash TEXT
) RETURNS TEXT AS $$
BEGIN
    RETURN encode(
        sha256(
            (p_action::TEXT || '|' || 
             p_resource_type || '|' || 
             COALESCE(p_resource_id::TEXT, '') || '|' || 
             COALESCE(p_metadata::TEXT, '{}') || '|' || 
             COALESCE(p_previous_hash, 'GENESIS'))::BYTEA
        ),
        'hex'
    );
END;
$$ LANGUAGE plpgsql IMMUTABLE;

-- Trigger para mantener cadena de hashes
CREATE OR REPLACE FUNCTION audit_log_chain() RETURNS TRIGGER AS $$
DECLARE
    v_previous_hash TEXT;
    v_new_hash TEXT;
BEGIN
    -- Obtener hash del último registro
    SELECT payload_hash INTO v_previous_hash
    FROM audit_log
    ORDER BY id DESC
    LIMIT 1;
    
    -- Calcular nuevo hash
    v_new_hash := calculate_audit_hash(
        NEW.action,
        NEW.resource_type,
        NEW.resource_id,
        NEW.metadata,
        v_previous_hash
    );
    
    NEW.payload_hash := v_new_hash;
    NEW.previous_hash := v_previous_hash;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_user_usage(
    p_user_id UUID,
    p_period_start TIMESTAMPTZ,
    p_period_end TIMESTAMPTZ
) RETURNS TABLE(
    event_type TEXT,
    count BIGINT,
    total_tokens BIGINT,
    total_cost_cents BIGINT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        um.event_type,
        COUNT(*)::BIGINT,
        COALESCE(SUM(um.tokens_used), 0)::BIGINT,
        COALESCE(SUM(um.cost_cents), 0)::BIGINT
    FROM usage_metrics um
    WHERE um.user_id = p_user_id
      AND um.created_at >= p_period_start
      AND um.created_at < p_period_end
    GROUP BY um.event_type;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### 4.3 Stripe Webhook Handler (TypeScript)
```typescript
// apps/api/src/webhooks/stripe.ts
import { NextRequest, NextResponse } from 'next/server';
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: '2024-11-20.acacia',
});

const supabase = createClient(
  process.env.SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!
);

export async function POST(request: NextRequest) {
  const body = await request.text();
  const signature = request.headers.get('stripe-signature')!;

  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(
      body,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET!
    );
  } catch (err: any) {
    console.error('Webhook signature verification failed:', err.message);
    return NextResponse.json({ error: 'Invalid signature' }, { status: 400 });
  }

  // Idempotency: verificar si ya procesamos este evento
  const { data: existing } = await supabase
    .from('webhook_events')
    .select('id')
    .eq('stripe_event_id', event.id)
    .single();

  if (existing) {
    return NextResponse.json({ received: true, duplicate: true });
  }

  try {
    switch (event.type) {
      case 'customer.subscription.created':
      case 'customer.subscription.updated':
        await handleSubscriptionUpsert(event.data.object as Stripe.Subscription);
        break;

      case 'customer.subscription.deleted':
        await handleSubscriptionCanceled(event.data.object as Stripe.Subscription);
        break;

      case 'invoice.payment_succeeded':
        await handlePaymentSucceeded(event.data.object as Stripe.Invoice);
        break;

      case 'invoice.payment_failed':
        await handlePaymentFailed(event.data.object as Stripe.Invoice);
        break;

      case 'checkout.session.completed':
        await handleCheckoutCompleted(event.data.object as Stripe.Checkout.Session);
        break;

      case 'customer.created':
        await handleCustomerCreated(event.data.object as Stripe.Customer);
        break;

      default:
        console.log(`Unhandled event type: ${event.type}`);
    }

    // Registrar evento procesado
    await supabase.from('webhook_events').insert({
      stripe_event_id: event.id,
      event_type: event.type,
      payload: event.data.object,
      processed_at: new Date().toISOString(),
    });

    return NextResponse.json({ received: true });
  } catch (error: any) {
    console.error(`Error processing ${event.type}:`, error);
    
    // Registrar fallo para retry
    await supabase.from('webhook_events').insert({
      stripe_event_id: event.id,
      event_type: event.type,
      payload: event.data.object,
      error: error.message,
      processed_at: new Date().toISOString(),
    });

    return NextResponse.json({ error: 'Processing failed' }, { status: 500 });
  }
}

// ─── HANDLERS ESPECÍFICOS ───

async function handleSubscriptionUpsert(subscription: Stripe.Subscription) {
  const customerId = subscription.customer as string;
  
  // Buscar usuario por stripe_customer_id
  const { data: user } = await supabase
    .from('users')
    .select('id')
    .eq('stripe_customer_id', customerId)
    .single();

  if (!user) {
    console.warn(`No user found for Stripe customer: ${customerId}`);
    return;
  }

  const tier = mapPriceToTier(subscription.items.data[0]?.price.id);
  const status = mapStatus(subscription.status);

  await supabase
    .from('subscriptions')
    .upsert({
      user_id: user.id,
      stripe_customer_id: customerId,
      stripe_subscription_id: subscription.id,
      tier,
      status,
      current_period_start: new Date(subscription.current_period_start * 1000).toISOString(),
      current_period_end: new Date(subscription.current_period_end * 1000).toISOString(),
      cancel_at: subscription.cancel_at 
        ? new Date(subscription.cancel_at * 1000).toISOString() 
        : null,
      canceled_at: subscription.canceled_at 
        ? new Date(subscription.canceled_at * 1000).toISOString() 
        : null,
      metadata: subscription.metadata,
    }, {
      onConflict: 'stripe_subscription_id'
    });

  // Registrar en auditoría
  await supabase.from('audit_log').insert({
    user_id: user.id,
    action: 'update',
    resource_type: 'subscription',
    metadata: { tier, status, subscription_id: subscription.id },
  });
}

async function handleSubscriptionCanceled(subscription: Stripe.Subscription) {
  await supabase
    .from('subscriptions')
    .update({
      status: 'canceled',
      canceled_at: new Date().toISOString(),
    })
    .eq('stripe_subscription_id', subscription.id);
}

async function handlePaymentSucceeded(invoice: Stripe.Invoice) {
  const customerId = invoice.customer as string;
  const amountPaid = invoice.amount_paid; // en centésimos

  const { data: user } = await supabase
    .from('users')
    .select('id')
    .eq('stripe_customer_id', customerId)
    .single();

  if (user) {
    await supabase.from('usage_metrics').insert({
      user_id: user.id,
      event_type: 'payment',
      event_data: {
        invoice_id: invoice.id,
        amount_cents: amountPaid,
        currency: invoice.currency,
      },
      cost_cents: amountPaid,
    });
  }
}

async function handlePaymentFailed(invoice: Stripe.Invoice) {
  const customerId = invoice.customer as string;

  const { data: user } = await supabase
    .from('users')
    .select('id, email')
    .eq('stripe_customer_id', customerId)
    .single();

  if (user) {
    // Enviar email de notificación
    await sendDunningEmail(user.email, invoice);

    // Actualizar estado de suscripción
    await supabase
      .from('subscriptions')
      .update({ status: 'past_due' })
      .eq('stripe_customer_id', customerId);
  }
}

async function handleCheckoutCompleted(session: Stripe.Checkout.Session) {
  const customerId = session.customer as string;
  const userId = session.metadata?.user_id;

  if (!userId) {
    console.warn('No user_id in checkout metadata');
    return;
  }

  // Vincular Stripe customer al usuario
  await supabase
    .from('users')
    .update({ stripe_customer_id: customerId })
    .eq('id', userId);

  // Activar suscripción si existe
  if (session.subscription) {
    const subscription = await stripe.subscriptions.retrieve(
      session.subscription as string
    );
    await handleSubscriptionUpsert(subscription);
  }
}

async function handleCustomerCreated(customer: Stripe.Customer) {
  // Opcional: crear registro de cliente
  console.log(`New Stripe customer: ${customer.id}`);
}

// ─── HELPERS ───

function mapPriceToTier(priceId: string | undefined): string {
  if (!priceId) return 'free';
  const tierMap: Record<string, string> = {
    [process.env.STRIPE_PRICE_STARTER!]: 'starter',
    [process.env.STRIPE_PRICE_PRO!]: 'pro',
    [process.env.STRIPE_PRICE_ENTERPRISE!]: 'enterprise',
  };
  return tierMap[priceId] || 'free';
}

function mapStatus(stripeStatus: string): string {
  const statusMap: Record<string, string> = {
    active: 'active',
    past_due: 'past_due',
    canceled: 'canceled',
    trialing: 'trialing',
    incomplete: 'past_due',
    incomplete_expired: 'canceled',
    unpaid: 'past_due',
  };
  return statusMap[stripeStatus] || 'active';
}

}
```
